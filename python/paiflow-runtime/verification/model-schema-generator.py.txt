import copy
import json
from pathlib import Path

contracts = Path('../../contracts/paiflow')
worker = 'worker.schema.json'
text = {'type': 'string', 'minLength': 1, 'maxLength': 256}
states = ['RUNNING', 'COMPLETED', 'FAILED', 'CANCELLED', 'EFFECT_UNKNOWN']
kinds = ['provider', 'authorization', 'protocol', 'transport', 'deadline']

def obj(properties, required=None, **extra):
    return {'type': 'object', 'additionalProperties': False,
            'required': list(properties) if required is None else required,
            'properties': properties, **extra}

def ref(name):
    return {'$ref': '#/$defs/' + name}

binding = {
    'runId': {'$ref': worker + '#/properties/runId'},
    'leaseId': {'$ref': worker + '#/properties/leaseId'},
    'attempt': {'$ref': worker + '#/properties/attempt'},
    'nodeId': text,
    'iterationId': {'anyOf': [{'type': 'null'}, text]},
    'workflowVersionId': {'$ref': worker + '#/properties/workflowVersionId'},
    'dslHash': {'$ref': worker + '#/properties/dslHash'},
    'operationId': {'type': 'string', 'minLength': 1, 'maxLength': 128},
    'resourceId': text,
    'resourceVersion': text,
    'provider': {'enum': ['xinghuo', 'openai']}}

def bound(properties, *, formal=False, required=None, **extra):
    fields = {k: {'$ref': '#/$defs/PrivateBinding/properties/' + k} for k in binding}
    if formal:
        fields.update(provider={'const': 'openai'}, iterationId={'type': 'null'},
                      nodeId={'type': 'string', 'pattern': '^spark-llm::.+$', 'maxLength': 256})
    fields.update(properties)
    return obj(fields, list(fields) if required is None else list(binding) + required, **extra)

    'topK': {'type': 'integer'},

    'functions': {'type': 'array', 'items': {'type': 'object'}},
    'timeoutSeconds': {'type': 'number', 'exclusiveMinimum': 0, 'maximum': 600}},
    ['resourceId', 'resourceVersion', 'provider', 'modelName', 'mode', 'messages', 'options'],
    **{'x-status': 'target_contract_not_enabled', 'allOf': [{
        'if': {'properties': {'mode': {'const': 'function_call'}}},
        'then': {'required': ['functions'], 'properties': {'provider': {'const': 'xinghuo'},
                 'options': {'required': ['questionType', 'functionChoice']}}}}]})
chat = copy.deepcopy(parameters)

chat['properties']['provider'] = {'const': 'openai'}
chat['properties']['mode'] = {'const': 'chat'}
chat['properties'].pop('functions')
chat['properties']['options'] = obj({k: options[k] for k in option_required})
chat.pop('allOf')

snapshot = obj({'kind': {'const': 'model'}, 'provider': binding['provider'],
    'version': text, 'modelName': text, 'enabled': {'const': True},
    'capabilities': {'type': 'array', 'uniqueItems': True, 'items': {'enum': ['chat', 'function_call']}}},
    ['kind', 'provider', 'version', 'modelName', 'enabled'], **{'x-status': 'target_contract_not_enabled'})
formal_snapshot = copy.deepcopy(snapshot)
formal_snapshot['required'].append('capabilities')
formal_snapshot['properties']['provider'] = {'const': 'openai'}
formal_snapshot['properties']['capabilities'] = {'const': ['chat']}
formal_snapshot['x-status'] = 'current_http_contract_requires_real_admission_ports'

request_props = {
    'leaseId': binding['leaseId'], 'attempt': binding['attempt'],
    'nodeId': text, 'iterationId': binding['iterationId'],
    'operation': {'const': 'model.invoke'}, 'parameters': ref('ModelParameters'),
    'idempotencyKey': {'type': 'string', 'minLength': 1, 'maxLength': 4096,
                       'pattern': '^[^\\u0000-\\u001f\\u007f]+$'}}
operation = obj(request_props, **{'x-status': 'target_contract_not_enabled', 'x-max-json-utf8-bytes': 2097152})
formal_operation = copy.deepcopy(operation)

    nodeId={'type': 'string', 'pattern': '^spark-llm::.+$', 'maxLength': 256})
formal_operation['x-status'] = 'current_http_contract_requires_real_admission_ports'

    'nextCursor': {'type': 'integer', 'minimum': 0, 'maximum': 10000},
    'done': {'type': 'boolean'}, 'lastSeq': {'type': 'integer', 'minimum': 0, 'maximum': 10000},

     'then': {'properties': {'operationState': {'const': 'COMPLETED'}, 'done': {'const': True},
                            'finishSeq': {'type': 'integer', 'minimum': 1}, 'failureKind': {'type': 'null'}}}}]

    'OpenAIChatStartAck': bound({'state': {'const': 'RUNNING'}}, formal=True),
    'FramesRequest': obj({'leaseId': binding['leaseId'], 'attempt': binding['attempt'],

    'FramesAck': bound(frame_fields, **{'allOf': frame_rules, 'x-status': 'target_contract_not_enabled'}),
    'OpenAIChatFramesAck': bound(frame_fields, formal=True, **{'allOf': frame_rules}),
    'CancelAck': bound({'state': {'enum': states}, 'stopped': {'type': 'boolean'},
        'proofId': {'anyOf': [{'type': 'null'}, text]}}, **{'allOf': [{
        'if': {'properties': {'stopped': {'const': True}}}, 'then': {'properties': {
            'state': {'enum': ['COMPLETED', 'FAILED', 'CANCELLED']}, 'proofId': text}}}]}),
    'PrivateParameters': obj({'binding': ref('PrivateBinding'),
        'parametersCanonicalBase64': {'type': 'string', 'minLength': 1, 'contentEncoding': 'base64'},
        'parametersSha256': binding['dslHash'],
        'optionsCanonicalBase64': {'type': 'string', 'minLength': 1, 'contentEncoding': 'base64'},
        'deadline': {'$ref': worker + '#/properties/deadline'}},
        **{'x-private-internal-bridge-only': True,
           'description': '中心实际WriteMapNull规范字节；解码原字节核SHA256，保全null与缺字段差异。没有credential/endpoint字段或默认桥。'})}
defs['OpenAIChatCancelAck'] = copy.deepcopy(defs['CancelAck'])

    iterationId={'type': 'null'}, nodeId={'type': 'string', 'pattern': '^spark-llm::.+$', 'maxLength': 256})

api_path = contracts/'internal.openapi.json'
api = json.loads(api_path.read_text(encoding='utf-8-sig'))
def add_path(path, operation_id, request, ack, description, parameters):
    api['paths'][path] = {'parameters': parameters, 'post': {'operationId': operation_id,
        'description': description, 'requestBody': {'required': True, 'content': {'application/json': {
            'schema': {'$ref': 'model-operations.schema.json#/$defs/'+request}}}},
        'responses': {'200': {'description': 'AjaxResult code=200/data，完整绑定及停止/尾帧仍须严格核对',
            'content': {'application/json': {'schema': obj({'code': {'const': 200}, 'msg': {'type': 'string'},
                'data': {'$ref': 'model-operations.schema.json#/$defs/'+ack}}, ['code', 'data'])}}},
            '400': {'description': '未知字段/非法类型/游标/参数'}, '401': {'description': '独立服务身份失效'},
            '403': {'description': '平台/空间/资源/费用授权拒绝'}, '409': {'description': '租约/固定版本/幂等/CAS冲突'},
            '410': {'description': '已到期/取消/失租约'}, '413': {'description': '真实UTF8正文预算超限'},
            '503': {'description': '缺真实资源/动态消息/审批/Producer受控transport桥，零operation写'}}}}
run_param = {'$ref': '#/components/parameters/RunId'}
resource_param = {'name': 'resourceId', 'in': 'path', 'required': True, 'schema': text}
op_param = {'name': 'operationId', 'in': 'path', 'required': True, 'schema': binding['operationId']}
prefix = '/internal/paiflow/v1/runs/{runId}'
add_path(prefix+'/resources/{resourceId}', 'modelResource', 'LeaseRequest', 'OpenAIChatSnapshot',
    '当前模型资源仅公开六字段，按当前租约及固定版本授权；私有endpoint/引用不出现在assignment或本响应。其他资源种类由所属能力另行实现。', [run_param, resource_param])
add_path(prefix+'/operations', 'invokeModel', 'OpenAIChatOperationRequest', 'OpenAIChatStartAck',
    '本批model.invoke仅Python顶层spark-llm/OpenAI chat，iterationId必null。真实运行已begin、权限/资源/动态消息/审批/受控transport全就绪后才可保存操作；其他能力仍有各自未接契约。', [run_param])
add_path(prefix+'/operations/{operationId}/frames', 'modelFrames', 'FramesRequest', 'OpenAIChatFramesAck',
    '只读正式原帧；分页交付投影及operationState分开，真实finish/lastSeq保全尾usage，支持空最终ACK。读取不新发Provider。', [run_param, op_param])
add_path(prefix+'/operations/{operationId}/cancel', 'cancelModel', 'LeaseRequest', 'OpenAIChatCancelAck',
    '请求停止后回真实事实；已派发无Provider proof仍stopped=false/EFFECT_UNKNOWN，不以本地关闭假称停止。', [run_param, op_param])
api['info']['description'] += ' 模型逐次操作共享契约已保存；正式能力仍依赖真实固定资源、动态消息授权、审批和Producer内部受控桥，缺任何项503/零operation写；没有新增私有Grant HTTP路由。'
api_path.write_text(json.dumps(api, ensure_ascii=False, indent=2)+'\n', encoding='utf-8')
